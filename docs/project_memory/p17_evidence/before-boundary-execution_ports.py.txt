"""Host-neutral ports for already formed structured actions only."""
from typing import Protocol
from continuity_engine.domain.action_capability import ActionReceipt, InternalActionRequest, ReceiptQuery
from continuity_engine.domain.execution import BlastRadius, WorldCapability


class WorldAdapter(Protocol):
    adapter_id: str
    def query(self, request: InternalActionRequest) -> ActionReceipt | ReceiptQuery: ...
    def execute(self, request: InternalActionRequest) -> ActionReceipt: ...
    def read_result(self, request: InternalActionRequest) -> dict: ...


class RealityBoundary(Protocol):
    def ready(self, route: WorldCapability) -> bool: ...
    def recoverable(self, route: WorldCapability) -> bool: ...
    def authorize(self, route: WorldCapability, request: InternalActionRequest, *, purpose: str) -> bool: ...
    def capacity(self, route: WorldCapability, request: InternalActionRequest, limits: BlastRadius) -> bool: ...


class ExecutionCredentialBroker(Protocol):
    def authorize_reference(self, route: WorldCapability, *, at, purpose: str) -> bool: ...
    def material_allowed(self, route: WorldCapability, value) -> bool: ...
